% Custom title page command
\newcommand{\covertitlepage}[6]{
  \thispagestyle{empty}
  
  % Full page background
  \begin{tikzpicture}[remember picture, overlay]
    % Background gradient
    \shade[bottom coverbg!100!coverdark, top coverbg!95!coverdark] 
      (current page.south west) rectangle (current page.north east);
    
    % Decorative border
    \draw[line width=2pt, color=covergold, rounded corners=8pt]
      ([xshift=0.6in, yshift=0.6in]current page.south west) --
      ([xshift=0.6in, yshift=0.6in]current page.north east) --
      ([xshift=-0.6in, yshift=0.6in]current page.north west) --
      ([xshift=-0.6in, yshift=-0.6in]current page.south west) -- cycle;
    
    % Inner decorative line
    \draw[line width=0.5pt, color=covergold!40, rounded corners=5pt]
      ([xshift=0.7in, yshift=0.7in]current page.south west) --
      ([xshift=0.7in, yshift=0.7in]current page.north east) --
      ([xshift=-0.7in, yshift=0.7in]current page.north west) --
      ([xshift=-0.7in, yshift=-0.7in]current page.south west) -- cycle;
    
    % Top decorative element - series banner
    \fill[coveraccent!85!black] 
      ([xshift=1.2in]current page.north west) rectangle 
      ([xshift=-1.2in, yshift=-0.4in]current page.north east);
    
    % Bottom decorative element
    \fill[coverdark!80!black] 
      ([xshift=1.2in]current page.south west) rectangle 
      ([xshift=-1.2in, yshift=0.4in]current page.south east);
    
    % Decorative horizontal lines
    \draw[line width=1pt, color=covergold]
      ([xshift=1.5in, yshift=-0.8in]current page.north west) --
      ([xshift=-1.5in, yshift=-0.8in]current page.north east);
    
    \draw[line width=1pt, color=covergold]
      ([xshift=1.5in, yshift=1.2in]current page.south west) --
      ([xshift=-1.5in, yshift=1.2in]current page.south east);
    
    % Corner ornaments - top left
    \draw[line width=3pt, color=coveraccent]
      ([xshift=0.8in, yshift=0.5in]current page.north west) --
      ([xshift=0.8in, yshift=0.8in]current page.north west) --
      ([xshift=1.1in, yshift=0.8in]current page.north west);
    
    % Corner ornaments - top right
    \draw[line width=3pt, color=coveraccent]
      ([xshift=-0.8in, yshift=0.5in]current page.north east) --
      ([xshift=-0.8in, yshift=0.8in]current page.north east) --
      ([xshift=-1.1in, yshift=0.8in]current page.north east);
    
    % Corner ornaments - bottom left
    \draw[line width=3pt, color=coveraccent]
      ([xshift=0.8in, yshift=-0.5in]current page.south west) --
      ([xshift=0.8in, yshift=-0.8in]current page.south west) --
      ([xshift=1.1in, yshift=-0.8in]current page.south west);
    
    % Corner ornaments - bottom right
    \draw[line width=3pt, color=coveraccent]
      ([xshift=-0.8in, yshift=-0.5in]current page.south east) --
      ([xshift=-0.8in, yshift=-0.8in]current page.south east) --
      ([xshift=-1.1in, yshift=-0.8in]current page.south east);

    \fill[covergold] ([xshift=0in, yshift=-0.8in]current page.north west) circle (2pt);
    \fill[covergold] ([xshift=0in, yshift=-0.8in]current page.north east) circle (2pt);

  \end{tikzpicture}
  
  % Content positioning using tikz for precise placement
  \begin{tikzpicture}[remember picture, overlay]
    
    % Series title at top
    \node[anchor=north] at ([xshift=0in, yshift=-1.2in]current page.north) {
      \parbox{5in}{
        \centering
        \color{coverlight}\fontsize{18pt}{22pt}\selectfont\textbf{#1}
      }
    };
    
    % Main title (story title) - center of page
    \node[anchor=center] at ([xshift=0in, yshift=-0.3in]current page.center) {
      \parbox{6in}{
        \centering
        \color{coverlight}\Huge\textbf{\textit{#4}}
      }
    };
    
    % Accent line under title
    \node[anchor=south] at ([xshift=0in, yshift=-1.1in]current page.center) {
      \parbox{3in}{
        \centering
        \color{covergold}\rule{3in}{1pt}
      }
    };
    
    % Issue number
    \node[anchor=north] at ([xshift=0in, yshift=0.6in]current page.center) {

      }

    \node[anchor=north] at ([xshift=0in, yshift=1.0in]current page.center) {
      \parbox{4in}{
        \centering
        \color{covergold}\large#3
      }
    };
    
    % Author
    \node[anchor=south] at ([xshift=0in, yshift=1.2in]current page.center) {
      \parbox{4in}{
        \centering
        \color{coverlight}\large#5
      }
    };
    
    % Copyright at bottom
    \node[anchor=south] at ([xshift=0in, yshift=0.8in]current page.south) {
      \parbox{6in}{
        \centering
        \color{covergold!70}\small#6
      }
    };
    
  \end{tikzpicture}
  
  % Force full page
  \clearpage
}
