\makeatletter
\def\@schapter#1{}% Hide unnumbered chapters completely
\def\@chapter[#1]#2{\ifnum \c@secnumdepth >\m@ne
  % Skip chapter numbering
  \fi
  \refstepcounter{chapter}%
  \addcontentsline{toc}{chapter}{\protect\numberline{}Chapter}%
  \chaptermark{#1}%
  \addtocontents{lof}{\protect\addvspace{10\p@}}%
  \addtocontents{lot}{\protect\addvspace{10\p@}}%
  \if@twocolumn
    \@topnewpage[\@makechapterhead{#1}]%
  \else
    \@makechapterhead{#1}%
    \@afterheading
  \fi}
\def\@makechapterhead#1{%
  % Just output nothing - suppress all chapter headings
}

% Title page generation command - clean black-and-white design
\newcommand{\maketitlepage}{
  \thispagestyle{empty}
  \setlength{\parindent}{0pt}
  \setlength{\parskip}{0pt}
  
  \begin{tikzpicture}[remember picture, overlay]
    
    
    
    % === CONTENT ELEMENTS (centered vertical layout) ===
    
    % Series title at top (escape & for TikZ node context)
    \node[anchor=north] at ([xshift=0in, yshift=-0.6in]current page.north) {
      \parbox{7in}{
        \centering
        \color{black}\Large\textbf{Cat \& Company}
      }
    };
    
    
    % Issue number
    \node[anchor=north] at ([xshift=0in, yshift=1.7in]current page.center) {
      \parbox{6in}{
        \centering
        \large\textbf{\issueNumber}
      }
    };
    
    % Page count
    \node[anchor=north] at ([xshift=0in, yshift=1.2in]current page.center) {
      \parbox{6in}{
        \centering
        \textit{\pageCount}
      }
    };
    
    
    % Story title (center of page - most prominent, large italic)
    \node[anchor=center] at ([xshift=0in, yshift=-0.9in]current page.center) {
      \parbox{7in}{
        \centering
        \color{black}\Huge\textbf{\textit{\storyTitle}}
      }
    };
    
    
    % Author name
    \node[anchor=south] at ([xshift=0in, yshift=-1.9in]current page.center) {
      \parbox{6in}{
        \centering
        \large\authorName
      }
    };
    
    % Copyright at bottom
    \node[anchor=north] at ([xshift=0in, yshift=0.6in]current page.south) {
      \parbox{7in}{
        \centering
        \small\copyrightLine
      }
    };
    
  \end{tikzpicture}
  
  % End title page
  \clearpage
}
